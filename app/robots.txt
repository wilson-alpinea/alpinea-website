import { MetadataRoute } from "next";

export default function robots(): MetadataRoute.Robots {
  return {
    rules: {
      userAgent: "*",
      allow: "/",
      // Banco de conteúdo interno e CRM (ferramentas internas dos
      // colaboradores) — não devem ser indexados nem rastreados.
      disallow: ["/database", "/crm"],
    },

    sitemap: "https://www.alpinea.io/sitemap.xml",
  };
}
